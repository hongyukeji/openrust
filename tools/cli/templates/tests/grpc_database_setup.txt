    let Ok(admin_url) = std::env::var("OPENRUST_TEST_DATABASE_URL") else {
        eprintln!("skipping generated gRPC database test: OPENRUST_TEST_DATABASE_URL is unset");
        return Ok(());
    };
    let database =
        testing::TestDatabase::create::<openrust_migration::Migrator>(&admin_url).await?;
    std::fs::write(
        config_dir.join("database.toml"),
        format!("[database]\nurl = {:?}\n", database.url()),
    )?;
